static void PaletteWhiteOut(S1Context* ctx) {
    ctx->var.v_pfade_start = 0x003F;
    ctx->reg.d4 = 0x15;

    // mainloop
    do {
        ctx->var.v_vbla_routine = 0x12;
        WaitForVBla(ctx);
        WhiteOut_ToWhite(ctx);
        RunPLC(ctx);
        ctx->reg.d4--;
    } while (ctx->reg.d4 != 0)
}
